### 3. Hooks详解

```jsx
function useCustomHook() {
  const [state, setState] = useState(0);
  
  useEffect(() => {
    // 副作用处理
    return () => {
      // 清理函数
    };
  }, [state]);
  
  return [state, setState];
}
```

**Hooks规则：**
- 只能在函数组件顶层调用
- 不能在循环、条件、嵌套函数中调用
- 依赖数组要完整

## React 18并发渲染

### 渲染器-调度器-提交工作流

```javascript
// 简化的渲染流程
function render(element, container) {
  // 1. 调度阶段
  const workInProgress = createWorkInProgress(element);
  
  // 2. 渲染阶段（可中断）
  while (workInProgress !== null) {
    workInProgress = performUnitOfWork(workInProgress);
    
    // 检查是否需要让出控制权
    if (shouldYield()) {
      return workInProgress;
    }
  }
  
  // 3. 提交阶段（不可中断）
  commitRoot(container);
}

function performUnitOfWork(fiber) {
  // 处理当前fiber
  beginWork(fiber);
  
  if (fiber.child) {
    return fiber.child;
  }
  
  // 处理兄弟节点
  while (fiber) {
    completeWork(fiber);
    if (fiber.sibling) {
      return fiber.sibling;
    }
    fiber = fiber.return;
  }
  
  return null;
}
```

## 浏览器渲染机制

### 分层和栅格化

```javascript
// 浏览器渲染流程
function browserRender() {
  // 1. 解析HTML构建DOM树
  const domTree = parseHTML(html);
  
  // 2. 解析CSS构建CSSOM树
  const cssomTree = parseCSS(css);
  
  // 3. 合并DOM和CSSOM构建渲染树
  const renderTree = combine(domTree, cssomTree);
  
  // 4. 布局计算
  const layoutTree = layout(renderTree);
  
  // 5. 分层
  const layers = createLayers(layoutTree);
  
  // 6. 栅格化
  const tiles = rasterize(layers);
  
  // 7. 合成
  composite(tiles);
}
```

```javascript
// 栅格化算法
function rasterize(layer) {
  const tiles = [];
  
  // 将图层分割成瓦片
  for (let y = 0; y < layer.height; y += TILE_SIZE) {
    for (let x = 0; x < layer.width; x += TILE_SIZE) {
      const tile = createTile(layer, x, y);
      tiles.push(tile);
    }
  }
  
  return tiles;
}
```

## 热更新实现

### Vite HMR实现

```javascript
// Vite HMR客户端
class HMRClient {
  constructor() {
    this.socket = new WebSocket('ws://localhost:24678');
    this.modules = new Map();
  }
  
  // 模块更新处理
  handleUpdate(update) {
    const { path, timestamp, updates } = update;
    
    // 1. 获取模块
    const module = this.modules.get(path);
    
    // 2. 执行更新
    updates.forEach(update => {
      if (update.type === 'js-update') {
        this.updateJSModule(path, update);
      } else if (update.type === 'css-update') {
        this.updateCSSModule(path, update);
      }
    });
  }
  
  // JS模块更新
  updateJSModule(path, update) {
    // 1. 获取模块实例
    const module = this.modules.get(path);
    
    // 2. 执行dispose
    if (module.hot && module.hot.dispose) {
      module.hot.dispose();
    }
    
    // 3. 重新执行模块
    const newModule = this.executeModule(path);
    
    // 4. 执行accept
    if (newModule.hot && newModule.hot.accept) {
      newModule.hot.accept();
    }
  }
}
```

### Webpack HMR实现

```javascript
// Webpack HMR实现
class WebpackHMR {
  constructor() {
    this.hot = {
      accept: (deps, callback) => {
        // 注册热更新回调
        this.acceptCallbacks.push({ deps, callback });
      },
      
      dispose: (callback) => {
        // 注册清理回调
        this.disposeCallbacks.push(callback);
      }
    };
  }
  
  // 模块更新
  updateModule(moduleId, newModule) {
    // 1. 执行dispose
    this.disposeCallbacks.forEach(callback => callback());
    
    // 2. 替换模块
    this.modules.set(moduleId, newModule);
    
    // 3. 执行accept
    this.acceptCallbacks.forEach(({ deps, callback }) => {
      if (deps.includes(moduleId)) {
        callback();
      }
    });
  }
}
```
